# Content Signals Policy (https://contentsignals.org/)
# search=yes:    allow indexing for traditional search results
# ai-input=yes:  allow content as real-time AI input (RAG, AI overviews)
# ai-train=yes:  allow content for AI model training (lets ACG be named from model memory)

# ------------------------------------------------------------------------
# One group governs every crawler.
#
# Google's specification: "Only one applicable group is used for a crawler...
# User-agent-specific groups and the global * group are not combined."
# https://developers.google.com/search/docs/crawling-indexing/robots/robots_txt
#
# This file previously carried 32 named crawler groups that each contained
# nothing but "Allow: /". Because groups are never combined, every one of those
# crawlers -- GPTBot, ClaudeBot, PerplexityBot, Bingbot, CCBot, Google-Extended,
# Applebot and the rest -- matched its own group and was therefore exempt from
# all three Disallow rules below. They could crawl the dealer admin view and the
# JSON-LD fragment. The groups looked permissive and were actually a hole.
#
# Deleting them changes nothing about what is permitted, because "Allow: /" was
# already the default, and it restores the three Disallow rules for everyone.
#
# The allow-everything-that-cites posture is deliberate and is preserved: AI and
# search crawlers are welcome. Only Bytespider is refused, and that is a
# deliberate exception kept below.
#
# Content-Signal is [SPECULATIVE]. It is not a supported robots.txt field and no
# crawler is known to honour it. It is retained as a declaration of intent. Do
# not describe it as a control.
# ------------------------------------------------------------------------

User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=yes
Allow: /
# Internal admin & dealer-only views - no value to crawlers, contain no public content.
Disallow: /dealer/admin.html
Disallow: /dealer/thanks.html
# JSON-LD include fragment, not a page. Kept out of sitemaps. HTML stub would
# trip tn-claim-guard (excluded fragment must not look like a document).
Disallow: /services-schema-block.html

# ------------------------------------------------------------------------
# The one refusal. TikTok/Doubao's crawler takes content without citing it.
# ------------------------------------------------------------------------
User-agent: Bytespider
Disallow: /
